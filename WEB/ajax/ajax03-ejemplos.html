La primera aplicación AJAX (Hola Mundo)

Una aplicación JavaScript descarga un archivo del servidor y muestra su contenido sin recargar la página. Toda aplicación AJAX se compone de cuatro grandes bloques:

  1. Instanciar el objeto XMLHttpRequest (siempre primero).
  2. Preparar la función de respuesta (con onreadystatechange u onload).
  3. Realizar la petición al servidor con open() y send().
  4. Ejecutar la función de respuesta cuando llegan los datos.
<script>
function cargar_documento() {
  // 1. Instanciar el objeto
  var xhttp = new XMLHttpRequest();
  // 2. Preparar la función de respuesta
  xhttp.onreadystatechange = function() {
    // 4. Ejecutar la función de respuesta
    if (this.readyState == 4 && this.status == 200) {
      document.getElementById("demo").innerHTML = this.responseText;
    }
  };
  // 3. Realizar la petición al servidor
  xhttp.open("GET", "holamundo.txt", true);
  xhttp.send();
}
</script>

Ejemplo 1: un menú que carga contenido sin recargar

Se construye una página index.html con un menú; al hacer clic en cada enlace, una función de AJAX carga el contenido de otro archivo dentro de un <div>, sin volver a cargar el menú.

index.html

<!DOCTYPE html>
<html>
<head>
  <title>Asíncrono Javascript y XML</title>
  <script language="JavaScript" type="text/javascript" src="js/codigo.js"></script>
</head>
<body>
  <div id="menu">
    <a href="javascript:llamadasin('ejemplos/ej1.html', 'contenido');">Ejemplo 1</a>
    <a href="javascript:llamadasin('ejemplos/ej2.html', 'contenido');">Ejemplo 2</a>
  </div>
  <div id="contenido">
    <p>En este div se cargará asíncronamente el contenido del ejemplo</p>
  </div>
</body>
</html>

codigo.js — la función genérica llamadasin()

function llamadasin(url, contenedor){
  var pagina_requerida = false;
  // comprueba el tipo de navegador y crea el objeto
  if (window.XMLHttpRequest) {
    pagina_requerida = new XMLHttpRequest();
  } else if (window.ActiveXObject) {
    try {
      pagina_requerida = new ActiveXObject("Msxml2.XMLHTTP");
    } catch (e) {
      try {
        pagina_requerida = new ActiveXObject("Microsoft.XMLHTTP");
      } catch (e2) { /* navegador sin soporte */ }
    }
  }
  // prepara la función que pinta la página
  pagina_requerida.onreadystatechange = function(){
    pintapagina(pagina_requerida, contenedor);
  };
  // métodos open y send
  pagina_requerida.open('GET', url, true);
  pagina_requerida.send(null);
}

// función que presenta la información en el contenedor
function pintapagina(pagina_requerida, contenedor){
  if (pagina_requerida.readyState == 4 &&
      (pagina_requerida.status == 200 || window.location.href.indexOf("http") == -1)) {
    document.getElementById(contenedor).innerHTML = pagina_requerida.responseText;
  }
}
Leer el ejemplo: observá cómo la función llamadasin(url, contenedor) es adaptativa: recibe tanto el archivo a pedir como el div donde se vuelca el resultado, así sirve para muchos enlaces sin duplicar código.
