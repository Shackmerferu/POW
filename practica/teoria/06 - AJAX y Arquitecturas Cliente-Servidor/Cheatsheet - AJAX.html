1 · DOM: buscar, leer, crear, borrar

Seleccionar elementos y escribir resultados en la página. textContent siempre para datos externos; innerHTML solo con HTML propio.

Demo en vivo

el.classList.remove("d-none");
el.classList.toggle("active");
el.disabled = true;              // bloquear input/select/botón
el.required = true;              // validar por JS

li.textContent = "nuevo item";
document.getElementById("lista").appendChild(li);

const html = filas.join("");                     // array de strings → un string

Regla: si el dato viene de la API o del usuario → textContent o escapeHtml() antes de innerHTML. Nunca confíes y pegues innerHTML = dato directo (XSS).

8 · JSON (parsear / serializar)

/* texto  →  objeto (respuesta de la API) */
const datos = JSON.parse('{"status":200,"data":[1,2]}');

/* objeto  →  texto  (enviar a la API, guardar en storage) */
const texto = JSON.stringify({ usuario: "ana", password: "x" });

/* parseo TOLERANTE: si el PHP imprimió warnings ANTES del JSON */
function parseJson(texto) {
  if (!texto || !texto.trim()) return {};
  try { return JSON.parse(texto); }
  catch {
    const ini = texto.indexOf("{"), fin = texto.lastIndexOf("}");
    if (ini >= 0 && fin > ini) {
      try { return JSON.parse(texto.slice(ini, fin + 1)); } catch { /* abajo */ }
    }
    throw new Error("La API no devolvió JSON válido: " + texto.slice(0, 200));
  }
}

9 · AJAX 1 · XMLHttpRequest (el clásico)

Si el enunciado dice expresamente "usar XMLHttpRequest" o piden AJAX en su forma base, este es el patrón.

Demo: GET real a una API de prueba
—
/* ---------- GET ---------- */
const xhr = new XMLHttpRequest();
xhr.open("GET", "api/listar_productos.php?token=" + encodeURIComponent(token), true);
xhr.onload = function () {
  if (xhr.readyState === 4 && xhr.status === 200) {
    const datos = JSON.parse(xhr.responseText);
    console.log(datos.data);
  } else {
    console.error("HTTP " + xhr.status);
  }
};
xhr.onerror = function () { console.error("falló la conexión"); };
xhr.send();

/* ---------- POST urlencoded (API con extract($_REQUEST)) ---------- */
const xhr2 = new XMLHttpRequest();
xhr2.open("POST", "api/login.php", true);
xhr2.setRequestHeader("Content-Type", "application/x-www-form-urlencoded");
xhr2.onload = function () { console.log(JSON.parse(xhr2.responseText)); };
xhr2.send("usuario=ana&password=1234");

/* ---------- POST application/json (API que lee el body) ---------- */
const xhr3 = new XMLHttpRequest();
xhr3.open("POST", "api/algo.php", true);
xhr3.setRequestHeader("Content-Type", "application/json");
xhr3.send(JSON.stringify({ campo: "valor" }));

/* TIP: encodeURIComponent SIEMPRE al armar la query a mano
   (espacios, "&", acentos, etc. no rompen la URL) */

/* ---------- RECOMENDADO: versión segura con params + errores ---------- */
async function getJSON(url, params) {
  const destino = new URL(url);                  // arma query encodeada
  Object.entries(params || {}).forEach(([clave, valor]) => {
    if (valor !== undefined && valor !== null && valor !== "") {
      destino.searchParams.set(clave, String(valor));
    }
  });

  const response = await fetch(destino.toString(), {
    method: "GET",
    headers: { "Accept": "application/json" },
    cache: "no-store"
  });

  const datos = await response.json();   // o parseJson(await response.text())
  if (!response.ok) {
    throw new Error(datos.menssage || datos.mensaje || "Error HTTP " + response.status);
  }
  return datos;
}

/* ---------- USOS TÍPICOS ---------- */
const lista    = await getJSON("api/listar_productos.php", { token });       // con token
const persona  = await getJSON("api/personas_listar.php",  { id: 5 });       // por id
const ciudades = await getJSON("api/ciudades.php");                         // sin params

11 · AJAX 3 · fetch POST (login, altas, ediciones)

/* ---------- A · POST x-www-form-urlencoded (API con extract($_REQUEST)) ---------- */
async function postUrlEncoded(url, valores, params = {}) {
  const destino = new URL(url);
  Object.entries(params).forEach(([k, v]) => destino.searchParams.set(k, String(v)));

  const response = await fetch(destino.toString(), {
    method: "POST",
    headers: {
      "Content-Type": "application/x-www-form-urlencoded",
      "Accept": "application/json"
    },
    body: new URLSearchParams(valores).toString()   // clave=valor&clave2=valor2
  });
  return response.json();
}

/* usos típicos */
const login     = await postUrlEncoded("api/login.php",
                    { usuario: "ana", password: "1234" });

const guardado  = await postUrlEncoded("api/personas_guardar.php",
                    { dni: "30111222", nombre: "Ana", apellido: "Pérez",
                      direccion: "Calle 1", ciudad: "10", provincia: "5",
                      email: "ana@mail.com", telefono: "2954-123" });

const editado   = await postUrlEncoded("api/personas_guardar_editar.php",
                    { nombre: "Ana", /* resto de campos */ },
                    { id: 5 });                 // el id en la QUERY STRING

/* ---------- B · POST application/json (API REST que lee el body) ---------- */
const response = await fetch("api/usuarios.php", {
  method: "POST",
  headers: { "Content-Type": "application/json", "Accept": "application/json" },
  body: JSON.stringify({ usuario: "ana", password: "1234" })
});
const datos = await response.json();

13 · Mini-librería "App" (versión compacta de la del proyecto)

Pegala completa en tu app.js: trae sesión, GET/POST, alertas Botstrap, spinner y control de token. El API_BASE calcula solo la ruta hermana ../api/.

/* ================== js/config.js ================== */
const API_BASE = (() => {
  if (window.API_BASE_OVERRIDE) return window.API_BASE_OVERRIDE.replace(/\/+$/, "");
  if (window.location.protocol === "file:") return "http://localhost/termos/api";
  return new URL("../api/", window.location.href).href.replace(/\/+$/, "");
})();

const API = {
  login:            API_BASE + "/login.php",
  verificaToken:    API_BASE + "/verifica_token.php",
  listar:           API_BASE + "/listar_productos.php",
  registrar:        API_BASE + "/registrar_productos.php",
  personasCantidad: API_BASE + "/personas_cantidad.php",
  personasListar:   API_BASE + "/personas_listar.php",
  personasGuardar:  API_BASE + "/personas_guardar.php",
  personasEditar:   API_BASE + "/personas_guardar_editar.php",
  ciudades:         API_BASE + "/ciudades.php",
  imagenes:         API_BASE + "/imagenes"
};

/* ================== js/app.js ================== */
"use strict";

const App = (() => {

  /* ---- SESIÓN (sessionStorage) ---- */
  function getToken()   { return sessionStorage.getItem("termos_token"); }
  function getUsuario() {
    const g = sessionStorage.getItem("termos_usuario");
    if (!g) return null;
    try   { return JSON.parse(g); } catch { return null; }
  }
  function saveSession(token, usuario) {
    sessionStorage.setItem("termos_token", token);
    sessionStorage.setItem("termos_usuario", JSON.stringify(usuario || {}));
  }
  function clearSession() {
    sessionStorage.removeItem("termos_token");
    sessionStorage.removeItem("termos_usuario");
  }

  /* ---- SEGURIDAD / URLS ---- */
  function escapeHtml(v) {
    return String(v ?? "").replace(/[&<>"']/g, (c) => ({
      "&": "&amp;", "<": "&lt;", ">": "&gt;", '"': "&quot;", "'": "&#39;"
    })[c]);
  }
  function imageUrl(archivo) { return API.imagenes + "/" + encodeURIComponent(archivo); }
  function withParams(url, params) {
    const destino = new URL(url);
    Object.entries(params || {}).forEach(([k, v]) => {
      if (v !== undefined && v !== null && v !== "") destino.searchParams.set(k, String(v));
    });
    return destino.toString();
  }

  /* ---- PARSEO TOLERANTE DE JSON ---- */
  function parseJson(texto) {
    if (!texto || !texto.trim()) return {};
    try { return JSON.parse(texto); }
    catch {
      const ini = texto.indexOf("{"), fin = texto.lastIndexOf("}");
      if (ini >= 0 && fin > ini) {
        try { return JSON.parse(texto.slice(ini, fin + 1)); } catch { /* abajo */ }
      }
      const error = new Error("La API no devolvió JSON válido: " + texto.slice(0, 200));
      throw error;
    }
  }

  async function leerRespuesta(response) {
    const datos = parseJson(await response.text());
    if (!response.ok) {
      throw new Error(datos.menssage || datos.mensaje || ("Error HTTP " + response.status));
    }
    return datos;
  }

  /* ---- PETICIONES ---- */
  async function getJSON(url, params) {
    const response = await fetch(withParams(url, params), {
      method: "GET", headers: { "Accept": "application/json" }, cache: "no-store"
    });
    return leerRespuesta(response);
  }
  async function postForm(url, formData, params) {
    const response = await fetch(withParams(url, params), {
      method: "POST", headers: { "Accept": "application/json" }, body: formData
    });
    return leerRespuesta(response);
  }
  async function postUrlEncoded(url, valores, params) {
    return postForm(url, new URLSearchParams(valores), params);
  }

  /* ---- ALERTAS BOOTSTRAP ---- */
  function showAlert(contenedor, mensaje, tipo) {
    if (!contenedor) return;
    tipo = tipo || "success";
    contenedor.innerHTML =
      '<div class="alert alert-' + escapeHtml(tipo) + '">'
      + escapeHtml(mensaje) + '</div>';
  }
  function clearAlert(contenedor) { if (contenedor) contenedor.innerHTML = ""; }

  /* ---- BOTÓN CARGANDO (spinner) ---- */
  function setLoading(boton, cargando, texto) {
    texto = texto || "Procesando...";
    if (cargando) {
      boton.dataset.textoOriginal = boton.innerHTML;
      boton.disabled = true;
      boton.innerHTML = '<span class="spinner-border spinner-border-sm" aria-hidden="true"></span> '
                        + escapeHtml(texto);
    } else {
      boton.disabled = false;
      if (boton.dataset.textoOriginal) boton.innerHTML = boton.dataset.textoOriginal;
    }
  }

  /* ---- CONTROL DE SESIÓN ---- */
  async function verifyToken(token) {
    if (!token) return false;
    try {
      const d = await getJSON(API.verificaToken, { token: token });
      const presente = d.tpken_presente === true || d.token_presente === true;
      return d.status === 200 && presente;
    } catch { return false; }
  }
  function redirectToLogin() {
    if (!/index\.html?$/i.test(window.location.pathname)) {
      window.location.replace("index.html");
    }
  }
  async function requireSession() {
    const token = getToken();
    if (!token) { redirectToLogin(); return null; }
    const ok = await verifyToken(token);
    if (!ok)    { clearSession(); redirectToLogin(); return null; }
    return token;
  }

  return { getToken, getUsuario, saveSession, clearSession, escapeHtml, imageUrl,
           withParams, getJSON, postForm, postUrlEncoded, showAlert, clearAlert,
           setLoading, verifyToken, requireSession, redirectToLogin };
})();

RECETA 1 · Listado desde la API + render seguro

Traer un array, ordenarlo en el cliente, escapar, inyectar HTML y manejar estados cargando/vacío/error. Funciona igual para productos, personas, noticias, etc.

let productos = [];                          // estado en memoria
let productosPorId = new Map();

async function cargarTermos() {
  const estado = document.getElementById("estado-termos");
  const grid   = document.getElementById("grid-termos");

  /* 1. estado de carga */
  estado.classList.remove("d-none");
  estado.textContent = "Cargando termos...";
  grid.innerHTML = "";

  try {
    /* 2. pedir a la API con el token */
    const datos = await App.getJSON(API.listar, { token: token });

    if (datos.status !== 200) {
      const mensaje = datos.menssage || datos.mensaje || "No se pudo listar.";

      /* token inválido → sesión vencida → al login */
      if (/desautorizado|token/i.test(mensaje)) {
        App.clearSession();
        App.redirectToLogin();
        return;
      }
      /* "algo malo paso" = tabla vacía en la API del parcial: estado vacío */
      if (/algo malo/i.test(mensaje)) {
        estado.innerHTML = '<p class="mb-0">No hay datos cargados todavía.</p>';
        return;
      }
      throw new Error(mensaje);
    }

    /* 3. ordenar en el cliente (localeCompare es la clave) */
    productos = (Array.isArray(datos.data) ? datos.data : []).slice()
      .sort((a, b) => String(a.titulo ?? "").localeCompare(
                     String(b.titulo ?? ""), "es", { sensitivity: "base" }));

    productosPorId = new Map(productos.map((p) => [String(p.id), p]));

    /* 4. render: map + join + escapeHtml en CADA campo */
    if (productos.length === 0) {
      estado.innerHTML = '<p class="mb-0">No hay datos cargados todavía.</p>';
      return;
    }
    estado.classList.add("d-none");
    grid.innerHTML = productos.map(renderizarProducto).join("");

  } catch (error) {
    /* 5. error visible */
    estado.innerHTML =
      '<p class="text-danger mb-0">No se pudo cargar: '
      + App.escapeHtml(error.message) + '</p>';
  }
}

/* ---------- render de UN item (card Bootstrap) ---------- */
function renderizarProducto(p) {
  const id     = App.escapeHtml(p.id);
  const titulo = App.escapeHtml(p.titulo);
  const desc   = App.escapeHtml(p.descripcion || "");
  const src    = p.imagen ? App.escapeHtml(App.imageUrl(p.imagen)) : "";

  const imagen = src
    ? '<img src="' + src + '" class="card-img-top" alt="' + titulo + '">'
    : '<div class="text-center p-5 text-muted">Sin imagen</div>';

  return `
    <div class="col-12 col-sm-6 col-xl-4">
      <div class="card h-100">

        <div class="card-body">
          <h5 class="card-title">${titulo}</h5>
          <p class="card-text">${desc}</p>
          <button class="btn btn-outline-success" data-ver="${id}">Ver detalle</button>
        </div>
      </div>
    </div>`;
}

/* HTML mínimo asociado:

  <input type="text"     id="usuario"  class="form-control" required>
  <input type="password" id="password" class="form-control" required>
  <button type="submit"  id="btn-ingresar" class="btn btn-success w-100">Ingresar</button>
</form>
*/

document.addEventListener("DOMContentLoaded", () => {
  const form    = document.getElementById("form-login");
  const alertas = document.getElementById("login-alert");
  const boton   = document.getElementById("btn-ingresar");
  const pass    = document.getElementById("password");

  /* 1. Si ya hay sesión válida en esta pestaña, entrar directo */
  (async () => {
    const token = App.getToken();
    if (!token) return;
    App.setLoading(boton, true, "Verificando sesión...");
    if (await App.verifyToken(token)) {
      window.location.replace("panel.html");
      return;
    }
    App.clearSession();               // token vencido → limpiar
    App.setLoading(boton, false);
  })();

  /* 2. Mostrar/ocultar contraseña */
  document.getElementById("btn-ver")?.addEventListener("click", () => {
    pass.type = pass.type === "text" ? "password" : "text";
  });

  /* 3. Enviar el login */
  form.addEventListener("submit", async (evento) => {
    evento.preventDefault();
    App.clearAlert(alertas);

    if (!form.checkValidity()) {
      form.classList.add("was-validated");
      return;
    }

    App.setLoading(boton, true, "Ingresando...");

    try {
      /* POST urlencoded porque la API usa extract($_REQUEST) */
      const datos = await App.postUrlEncoded(API.login, {
        usuario:  form.elements.usuario.value.trim(),
        password: form.elements.password.value
      });

      if (datos.status !== 200 || !datos.token) {
        throw new Error(
          datos.menssage || datos.mensaje || "Usuario o contraseña incorrectos."
        );
      }

      /* 4. Guardar sesión y entrar al panel */
      App.saveSession(datos.token, {
        nombre:   datos.nombre,
        apellido: datos.apellido,
        id:       datos.id_usurio || datos.id_usuario || null   // typo de la API
      });
      window.location.replace("panel.html");

    } catch (error) {
      App.showAlert(alertas, error.message, "danger");
      App.setLoading(boton, false);
      pass.focus(); pass.select();
    }
  });
});

RECETA 4 · Búsqueda binaria por DNI sobre la API

Si la tabla va 1..cantidad ordenada por DNI y hay un endpoint "listar por ID", el personas_cantidad.php da el techo y personas_listar.php?id=N la fila N.

/* Devuelve la persona si existe el DNI, o null si no existe */
async function buscarPersonaPorDni(dniBuscado) {

  /* 1. techo de la búsqueda: cuántas filas hay */
  const cantidadRespuesta = await App.getJSON(API.personasCantidad);

  if (cantidadRespuesta.status !== 200) {
    /* tabla vacía → cualquier DNI es inexistente */
    if (/algo malo/i.test(cantidadRespuesta.menssage || "")) return null;
    throw new Error(cantidadRespuesta.menssage || "No se pudo obtener la cantidad.");
  }

  /* 2. búsqueda binaria clásica */
  let inicio = 1;
  let fin    = Number(cantidadRespuesta.cantidad);

  while (inicio <= fin) {
    const medio = Math.floor((inicio + fin) / 2);

    const respuesta = await App.getJSON(API.personasListar, { id: medio });

    if (respuesta.status !== 200 ||
        !Array.isArray(respuesta.persona) ||
        respuesta.persona.length === 0) {
      fin = medio - 1;                  // evita bucle si la fila no existe
      continue;
    }

    const persona  = respuesta.persona[0];
    const dniMedio = Number(persona.dni);

    if (dniMedio === dniBuscado) return persona;      // ENCONTRADO
    if (dniMedio < dniBuscado)   inicio = medio + 1;  // media < buscado → derecha
    else                         fin   = medio - 1;   // media > buscado → izquierda
  }

  return null;                          // no existe esa persona
}
