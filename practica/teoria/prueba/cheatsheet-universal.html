Cheatsheet Universal del Parcial POW

HTML + CSS + Bootstrap + JS + Fetch + API PHP dada — leyendo y aplicando, en orden.

Cómo funciona ESTE parcial: la API (api/*.php) y la BD (termos.sql) vienen dadas y NO se tocan. Vos construís el front/ (HTML + CSS + JS) que se conecta a esa API con fetch y un token.
Las 4 trampas de esta API (tenelas en la cabeza SIEMPRE):
  1. Los errores llegan con error HTTP 200, pero adentro del JSON viene status: 500. Nunca confíes solo en respuesta.ok: chequea resultado.status.
  2. El campo de error se llama menssage (con doble s, typo del enunciado).
  3. verifica_token.php responde tpken_presente; login devuelve id_usurio. Aceptá ambos nombres posibles.
  4. Tabla de productos vacía → status 500 con "algo malo paso": eso significa "listado vacío", no error.

6. Conectar el HTML con JavaScript

JS controla la página con tres herramientas: getElementById (leer/escribir), eventos (reaccionar) y innerHTML/value (escribir/leer). Estructura recomendada de tus .js (así es el app.js del proyecto resuelto):

// ============ js/app.js — helpers que usa TODA la página ============
"use strict";

/* --- Lectura/escritura de la sesión (token en sessionStorage) --- */
const App = {

    saveSession(token, usuario) {
        sessionStorage.setItem(APP_STORAGE.token, token);
        sessionStorage.setItem(APP_STORAGE.usuario, JSON.stringify(usuario ?? {}));
    },

    getToken()   { return sessionStorage.getItem(APP_STORAGE.token) ?? ""; },
    getUsuario() { try { return JSON.parse(sessionStorage.getItem(APP_STORAGE.usuario) ?? "null"); } catch { return null; } },

    clearSession() {
        sessionStorage.removeItem(APP_STORAGE.token);
        sessionStorage.removeItem(APP_STORAGE.usuario);
    },

    /* --- Alertas --- */
    showAlert(contenedor, mensaje, tipo = "danger") {
        const seguro = App.escapeHtml(mensaje);
        contenedor.innerHTML = `
            <div class="alert alert-${tipo} alert-dismissible fade show" role="alert" aria-live="polite">
                ${seguro}
                <button type="button" class="btn-close" data-bs-dismiss="alert"></button>
            </div>`;
    },

    /* --- Escapar TODO dato que llegue de la API o de un usuario (XSS) --- */
    escapeHtml(valor) {
        return String(valor ?? "").replace(/[&<>"']/g, (caracter) => {
            return { "&": "&amp;", "<": "&lt;", ">": "&gt;",
                     '"': "&quot;", "'": "&#39;" }[caracter];
        });
    },

    /* --- URL de imagen: API.imagenes + nombre del archivo --- */
    imageUrl(nombre) {
        const limpio = String(nombre ?? "").replace(/^\/+/, "");
        return `${API.imagenes}/${encodeURIComponent(limpio)}`;
    },

    /* --- Parser tolerante: la API a veces "contamina" el JSON ---- */
    parseJson(texto) {
        if (!texto || !texto.trim()) return {};
        try { return JSON.parse(texto); }
        catch {
            const inicio = texto.indexOf("{");
            const fin = texto.lastIndexOf("}");
            if (inicio >= 0 && fin > inicio) {
                return JSON.parse(texto.slice(inicio, fin + 1));
            }
            throw new Error("La API no devolvió JSON válido");
        }
    },

    /* --- Requerir sesión en páginas protegidas (panel.html) --- */
    async requireSession() {
        const token = App.getToken();
        if (!token) { window.location.replace("index.html"); return null; }

        try {
            const datos = await App.getJSON(API.verificaToken, { token });
            const presente = datos.tpken_presente === true || datos.token_presente === true;
            if (datos.status === 200 && presente) return token;
        } catch { /* sigue */ }

        App.clearSession();
        window.location.replace("index.html");
        return null;
    }
};

7. Fetch contra esta API (GET/POST + token)

Concepto AJAX: pedir datos al servidor sin recargar la página; PHP responde JSON; JS lo dibuja. En este parcial hay 3 formas de llamar (reutilizalas tal cual):

// ============ GET con parámetros en la query ============
async function getJSON(url, parametros = {}) {
    const query = new URLSearchParams(parametros);   // ?token=...&id=...
    const respuesta = await fetch(`${url}?${query}`);

    // IMPORTANTE: la API contesta HTTP 200 hasta en los errores.
    // Se lee SIEMPRE el status LÓGICO de adentro del JSON.
    return App.parseJson(await respuesta.text());
}

// ============ POST con campos de texto (x-www-form-urlencoded) ============
async function postUrlEncoded(url, campos = {}) {
    const cuerpo = new URLSearchParams(campos);
    const respuesta = await fetch(url, {
        method: "POST",
        headers: { "Content-Type": "application/x-www-form-urlencoded" },
        body: cuerpo
    });
    return App.parseJson(await respuesta.text());
}

// ============ POST con archivo (multipart/form-data) ============
async function postForm(url, formDatos, extra = {}) {
    if (extra.token) formDatos.append("token", extra.token);
    // NO configurar Content-Type a mano: el navegador arma el multipart
    // con su "boundary" y si lo pisás, se rompe.
    const respuesta = await fetch(url, { method: "POST", body: formDatos });
    return App.parseJson(await respuesta.text());
}
1 El chequeo de TODA respuesta: 
const resultado = await App.getJSON(url, { token });
if (resultado.status !== 200) {
    throw new Error(resultado.menssage || resultado.message || "Error desconocido");
}
// a partir de acá: resultado.data | resultado.persona | resultado.localidades | ...
  • respuesta.json() corta todo si el contenido no es JSON → por eso usamos parseJson tolerante.
  • resultado.menssage (doble s) es el nombre real de esta API. Probá ambos por las dudas.
  • Listado de productos vacío o falla de BD → status 500 → tratá como "sin productos" en vez de error fatal (a menos que el token esté mal: ahí sí salí a relogear).
2 Cuándo GET y cuándo POST: leer/listar/consultar = GET (parámetros en la URL). Enviar/insertar/actualizar/login/archivo = POST (datos en el body, el token viaja como campo aparte del body).

async function leerMetadatos(archivo) {
    const buffer = await archivo.arrayBuffer();
    const tags = ExifReader.load(buffer);

    const texto = (tag) => {
        if (!tag) return "";
        const valor = tag.description ?? tag.value;
        return String(valor ?? "").trim();
    };

    return {
        titulo:             texto(tags.Headline)          || texto(tags.title),
        titulo_imagen:      texto(tags.title)             || texto(tags.Headline),
        descripcion:        texto(tags["Caption/Abstract"]) || texto(tags.description),
        descripcion_larga:  texto(tags.ExtDescrAccessibility) || texto(tags.description),
        texto_alternativo:  texto(tags.AltTextAccessibility)  || texto(tags.Headline)
    };
}

9 Personas: búsqueda binaria por DNI (2:15–2:50). No existe "buscar por DNI" en la API → nosotros lo construimos: pedimos la cantidad y consultamos el id del medio hasta encontrar (la tabla está ordenada por id): 
async function buscarPersonaPorDni(dniBuscado) {
    const cantidadRespuesta = await App.getJSON(API.personasCantidad, { token });
    if (cantidadRespuesta.status !== 200) throw new Error("No se pudo obtener la cantidad");

    let inicio = 1;
    let fin = Number(cantidadRespuesta.cantidad);       // límite superior: 99 personas

    while (inicio <= fin) {
        const medio = Math.floor((inicio + fin) / 2);

        const respuesta = await App.getJSON(API.personasListar, { id: medio });
        if (respuesta.status !== 200 || !Array.isArray(respuesta.persona) || respuesta.persona.length === 0) {
            fin = medio - 1;
            continue;
        }

        const persona = respuesta.persona[0];
        const dniMedio = Number(persona.dni);           // convertir a Number: comparar tipos iguales

        if (dniMedio === dniBuscado) return persona;
        if (dniMedio < dniBuscado) inicio = medio + 1;  // el DNI buscado es mayor: vamos a la derecha
        else fin = medio - 1;                          // es menor: a la izquierda
    }
    return null;    // no encontrado -> MODO ALTA
}
